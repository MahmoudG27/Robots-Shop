# Install Prometheus Stack using Helm

# Adding the Repo
helm repo add prometheus-community https://prometheus-community.github.io/helm-charts
helm repo update

# Install the stack
helm install monitoring prometheus-community/kube-prometheus-stack \
  -n monitoring \
  --create-namespace \
  -f infra/monitoring/values.yaml

# Connect Grafana
kubectl get svc -n monitoring
--> http://localhost:30007
--> user: admin - pass: admin123

Cluster Dashboard in Grafana:

➜ Dashboards
➜ New
➜ New Dashboard
➜ Add Visualization

1- Choose Data Source = "Prometheus"

2- Adding the Queries (Cluster Overview)

1) CPU Usage (Cluster)
Panel Type: Time series
sum(rate(container_cpu_usage_seconds_total{container!=""}[5m]))

2) CPU Usage per Pod
Panel Type: Time series
sum by (pod) (
  rate(container_cpu_usage_seconds_total{container!=""}[5m])
)

3) Memory Usage (Cluster)
Panel Type: Time series
sum(container_memory_usage_bytes{container!=""})

4) Memory Usage per Pod
sum by (pod) (container_memory_usage_bytes{container!=""})

5) Node CPU Usage
100 - avg(rate(node_cpu_seconds_total{mode="idle"}[5m])) * 100

6) Node Memory Usage
(1 - (node_memory_MemAvailable_bytes / node_memory_MemTotal_bytes)) * 100

7) Network Receive
sum(rate(container_network_receive_bytes_total[5m]))

8) Network Transmit
sum(rate(container_network_transmit_bytes_total[5m]))

9) Pod Count (Running)
count(kube_pod_status_phase{phase="Running"})

10) Restart Count
sum(kube_pod_container_status_restarts_total)

11) CPU Requests vs Usage
sum(rate(container_cpu_usage_seconds_total[5m]))
/
sum(kube_pod_container_resource_requests_cpu_cores)

12) Memory Requests vs Usage
sum(kube_pod_container_resource_requests_memory_bytes)
/
sum(container_memory_usage_bytes)
